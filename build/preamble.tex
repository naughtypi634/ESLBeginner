% ---------- document title ----------
\newcommand{\esltitle}[1]{%
  \par\vspace{4pt}%
  {\fontsize{25pt}{30pt}\selectfont\bfseries #1}\par
  \vspace{18pt}
}

% ---------- pattern component marker (print-friendly: bold + light gray) ----------
\newcommand{\eslmark}[1]{\colorbox{filllight}{\textbf{#1}}}

% ---------- underline emphasis (定语从句练习) ----------
\newcommand{\esluline}[1]{\uline{#1}}

% ---------- section header ----------
\newcommand{\eslsection}[3]{%
  \par\vspace{14pt}%
  \noindent{\dispfont\bfseries\fontsize{16pt}{19pt}\selectfont #1}\hspace{10pt}%
  {\fontsize{14pt}{17pt}\selectfont\bfseries #2}%
  \if\relax\detokenize{#3}\relax\else\hspace{7pt}{\color{muted}\dispspaced\footnotesize #3}\fi\par
  \vspace{9pt}
}

% ---------- pattern formula ----------
\newcommand{\eslformula}[2]{%
  \par\vspace{6pt}%
  {\fontsize{11pt}{15pt}\selectfont\bfseries #1}\par
  \vspace{2pt}
  {\color{muted}\bodyfont\small #2}\par
  \vspace{8pt}
}
\newcommand{\eslformulasimple}[1]{\eslformula{#1}{}}

% ---------- note ----------
\newcommand{\eslnote}[1]{\par\noindent\begin{minipage}[t]{\linewidth}%
  {\color{muted}\small #1}\end{minipage}\par\vspace{8pt}}

% ---------- sub-header ----------
\newcommand{\eslsubheader}[2]{%
  \par\vspace{9pt}%
  \noindent{\bfseries\small #1}%
  \if\relax\detokenize{#2}\relax\else\hspace{7pt}{\color{muted}\dispspaced\footnotesize #2}\fi\par
  \vspace{4pt}
}

% ---------- en / cn sentence pair ----------
\newcommand{\eslpair}[2]{%
  \par\vspace{2pt}%
  {\bodyfont #1}\par
  \vspace{1.5pt}
  {\color{muted}\footnotesize #2}\par
  \vspace{4.5pt}
}

% ---------- word / translation / example row ----------
\newcommand{\esltriple}[3]{%
  \par\vspace{2pt}%
  \noindent\begin{tabular}{@{}>{\RaggedRight}p{0.16\textwidth}@{\hspace{9pt}}>{\RaggedRight}p{0.25\textwidth}@{\hspace{9pt}}>{\RaggedRight}p{0.47\textwidth}@{}}
    {\bfseries #1} & {\bfseries #2} & {\bodyfont #3}
  \end{tabular}\par
  \vspace{6pt}
}

% ---------- numbered exercise item ----------
\newcommand{\eslexitem}[3]{%
  \par\vspace{2pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}l@{\hspace{9pt}}X@{}}
    {\dispfont\bfseries\footnotesize #1} &
    \RaggedRight {\bfseries #2}\par\vspace{2pt}{\color{muted}\bodyfont\small #3}
  \end{tabularx}\par
  \vspace{6pt}
}

% ---------- compact numbered exercise item (定语从句练习) ----------
\newcommand{\eslcompactitem}[3]{%
  \par\vspace{1pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}l@{\hspace{7pt}}X@{}}
    {\dispfont\bfseries\footnotesize #1} &
    \RaggedRight {\bfseries\small #2}\par\vspace{1.5pt}{\color{muted}\bodyfont\footnotesize #3}
  \end{tabularx}\par
  \vspace{3.5pt}
}

% ---------- plain numbered question ----------
\newcommand{\eslqitem}[2]{%
  \par\vspace{2pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}l@{\hspace{9pt}}X@{}}
    {\dispfont\bfseries\footnotesize #1} &
    \RaggedRight\bodyfont #2
  \end{tabularx}\par
  \vspace{6pt}
}

% ---------- dash bullet item ----------
\newcommand{\esldash}[1]{%

  \vspace{6pt}
}
