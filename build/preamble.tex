% ---------- page ----------
\usepackage[a4paper, top=24mm, bottom=20mm, left=23mm, right=23mm]{geometry}

\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyfoot[C]{\color{faint}\dispfont\footnotesize\thepage}

% ---------- colors ----------
\usepackage[table]{xcolor}
\definecolor{accent}{HTML}{0E6E62}
\definecolor{accentdeep}{HTML}{0A524A}
\definecolor{accentsoft}{HTML}{F0F6F5}
\definecolor{ink}{HTML}{1C2329}
\definecolor{muted}{HTML}{5F6B78}
\definecolor{faint}{HTML}{9AA4AE}
\definecolor{hairline}{HTML}{E3E0D8}

% ---------- fonts ----------
\usepackage{fontspec}
\usepackage{xeCJK}
\setmainfont{Noto Sans}[Ligatures=TeX, AutoFakeBold=2.5, AutoFakeSlant=0.25]

\newfontfamily\bodyfont{Noto Sans}[Ligatures=TeX]
\newfontfamily\dispfont{Montserrat}[Ligatures=TeX]
\newfontfamily\dispspaced{Montserrat}[LetterSpace=8]

% ---------- layout ----------
\usepackage{tabularx}
\usepackage{ragged2e}
\usepackage{microtype}
\linespread{1.0}
\setlength{\baselineskip}{13.8pt}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% ============================================================
%  Component macros
% ============================================================

% ---------- document title ----------
\newcommand{\esltitle}[1]{%
  \par\vspace{4pt}%
  {\fontsize{23pt}{28pt}\selectfont\bfseries #1}\par
  \vspace{10pt}
  {\color{accent}\hrule height 2.2pt width 15mm}\par
  \vspace{3pt}
  {\color{hairline}\hrule height 0.5pt}\par
  \vspace{14pt}
}

% ---------- section header ----------
\newcommand{\eslsection}[3]{%
  \par\vspace{15pt}%
  \noindent{\color{accent}\dispfont\bfseries\large #1}\hspace{9pt}%
  {\fontsize{13.5pt}{16.5pt}\selectfont\bfseries #2}%
  \if\relax\detokenize{#3}\relax\else\hspace{7pt}{\color{muted}\dispspaced\footnotesize #3}\fi\par
  \vspace{4pt}
  {\color{hairline}\hrule height 0.5pt}\par
  \vspace{8pt}
}

% ---------- pattern formula ----------
% Typographic treatment — no box, no side bar, no label:
% the formula itself carries the accent (deep teal, semibold, larger).
\newcommand{\eslformula}[2]{%
  \par\vspace{7pt}%
  {\color{accentdeep}\fontsize{11pt}{15pt}\selectfont\bfseries #1}\par
  \vspace{2pt}
  {\color{muted}\bodyfont\small #2}\par
  \vspace{8pt}
}
\newcommand{\eslformulasimple}[1]{\eslformula{#1}{}}

% ---------- note ----------
\newcommand{\eslnote}[1]{\par{\color{muted}\small #1}\par\vspace{6pt}}

% ---------- sub-header ----------
\newcommand{\eslsubheader}[2]{%
  \par\vspace{9pt}%
  \noindent{\color{accent}\rule{5pt}{1pt}}\hspace{7pt}%
  {\bfseries\footnotesize #1}%
  \if\relax\detokenize{#2}\relax\else\hspace{7pt}{\color{muted}\dispspaced\footnotesize #2}\fi\par
  \vspace{5pt}
}

% ---------- en / cn sentence pair ----------
\newcommand{\eslpair}[2]{%
  \par\vspace{2pt}%
  {\bodyfont #1}\par
  \vspace{1.5pt}
  {\color{muted}\footnotesize #2}\par
  \vspace{2.5pt}
  {\color{hairline}\hrule height 0.3pt}\par
  \vspace{5pt}
}

% ---------- word / translation / example row ----------
\newcommand{\esltriple}[3]{%
  \par\vspace{2pt}%
  \noindent\begin{tabular}{@{}>{\RaggedRight}p{0.16\textwidth}@{\hspace{9pt}}>{\RaggedRight}p{0.25\textwidth}@{\hspace{9pt}}>{\RaggedRight}p{0.47\textwidth}@{}}
    {\bfseries #1} & {\color{accentdeep}\bodyfont #2} & {\bodyfont #3}
  \end{tabular}\par
  \vspace{2pt}
  {\color{hairline}\hrule height 0.3pt}\par
  \vspace{5pt}
}

% ---------- numbered exercise item ----------
\newcommand{\eslexitem}[3]{%
  \par\vspace{2pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}l@{\hspace{9pt}}X@{}}
    {\color{accent}\dispfont\bfseries\footnotesize #1} &
    \RaggedRight {\bfseries #2}\par\vspace{2pt}{\color{muted}\bodyfont\small #3}
  \end{tabularx}\par
  \vspace{2pt}
  {\color{hairline}\hrule height 0.3pt}\par
  \vspace{5pt}
}

% ---------- plain numbered question ----------
\newcommand{\eslqitem}[2]{%
  \par\vspace{2pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}l@{\hspace{9pt}}X@{}}
    {\color{accent}\dispfont\bfseries\footnotesize #1} &
    \RaggedRight\bodyfont #2
  \end{tabularx}\par
  \vspace{2pt}
  {\color{hairline}\hrule height 0.3pt}\par
  \vspace{5pt}
}

% ---------- dash bullet item ----------
\newcommand{\esldash}[1]{%
  \par\vspace{2pt}%
  \noindent{\color{accent}\dispfont\bfseries ---}\hspace{8pt}%
  \begin{minipage}[t]{\dimexpr\textwidth-26pt\relax}\bodyfont #1\end{minipage}\par
  \vspace{2pt}
  {\color{hairline}\hrule height 0.3pt}\par
  \vspace{5pt}
}

% ---------- soft accent tag ----------
\newcommand{\eslpattag}[1]{%
  \par\vspace{4pt}%
  \noindent\colorbox{accentsoft}{\strut\color{accentdeep}\bfseries\footnotesize #1}\par
  \vspace{6pt}
}

% ---------- rule head (comparatives) ----------
\newcommand{\eslrulehead}[2]{%
  \par\vspace{8pt}%
  \noindent{\color{accent}\dispfont\bfseries\footnotesize Rule #1}\hspace{9pt}%
  {\fontsize{11.5pt}{14pt}\selectfont\bfseries #2}\par
  \vspace{5pt}
}

% ---------- page break ----------
\newcommand{\eslpagebreak}{\clearpage}
