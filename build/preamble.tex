% ---------- page ----------
\usepackage[a4paper, top=26mm, bottom=22mm, left=23mm, right=23mm]{geometry}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyfoot[R]{\footnotesize\color{faint}\dispfont\thepage}

% ---------- colors (grayscale, print-friendly) ----------
\usepackage[table]{xcolor}
\definecolor{ink}{HTML}{1A1A1A}       % near-black body text
\definecolor{muted}{HTML}{595959}     % secondary text
\definecolor{faint}{HTML}{8C8C8C}     % tertiary / page number
\definecolor{hairline}{HTML}{D9D9D9}  % light gray table rules
\definecolor{filllight}{HTML}{F2F2F2} % light gray table header band
\arrayrulecolor{hairline}

% ---------- fonts ----------
\usepackage{fontspec}
\usepackage{xeCJK}
\setmainfont{Noto Sans}[Ligatures=TeX, AutoFakeBold=2.5, AutoFakeSlant=0.25]
\setCJKmainfont{Noto Sans SC}[AutoFakeBold=2.5]
\newfontfamily\bodyfont{Noto Sans}[Ligatures=TeX]
\newfontfamily\dispfont{Montserrat}[Ligatures=TeX]
\newfontfamily\dispspaced{Montserrat}[LetterSpace=8]

% ---------- layout ----------
\usepackage{tabularx}
\usepackage{ragged2e}
\usepackage{microtype}
\linespread{1.0}
\setlength{\baselineskip}{13.8pt}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% ============================================================
%  Component macros
% ============================================================

% ---------- document title ----------
\newcommand{\esltitle}[1]{%
  \par\vspace{4pt}%
  {\fontsize{25pt}{30pt}\selectfont\bfseries #1}\par
  \vspace{18pt}
}

% ---------- section header ----------
\newcommand{\eslsection}[3]{%
  \par\vspace{16pt}%
  \noindent{\dispfont\bfseries\large #1}\hspace{9pt}%
  {\fontsize{14pt}{17pt}\selectfont\bfseries #2}%
  \if\relax\detokenize{#3}\relax\else\hspace{7pt}{\color{muted}\dispspaced\footnotesize #3}\fi\par
  \vspace{8pt}
}

% ---------- pattern formula ----------
\newcommand{\eslformula}[2]{%
  \par\vspace{6pt}%
  {\fontsize{11pt}{15pt}\selectfont\bfseries #1}\par
  \vspace{2pt}
  {\color{muted}\bodyfont\small #2}\par
  \vspace{8pt}
}
\newcommand{\eslformulasimple}[1]{\eslformula{#1}{}}

% ---------- note ----------
\newcommand{\eslnote}[1]{\par{\color{muted}\small #1}\par\vspace{8pt}}

% ---------- sub-header ----------
\newcommand{\eslsubheader}[2]{%
  \par\vspace{9pt}%
  \noindent{\bfseries\small #1}%
  \if\relax\detokenize{#2}\relax\else\hspace{7pt}{\color{muted}\dispspaced\footnotesize #2}\fi\par
  \vspace{4pt}
}

% ---------- en / cn sentence pair ----------
\newcommand{\eslpair}[2]{%
  \par\vspace{2pt}%
  {\bodyfont #1}\par
  \vspace{1.5pt}
  {\color{muted}\footnotesize #2}\par
  \vspace{5.5pt}
}

% ---------- word / translation / example row ----------
\newcommand{\esltriple}[3]{%
  \par\vspace{2pt}%
  \noindent\begin{tabular}{@{}>{\RaggedRight}p{0.16\textwidth}@{\hspace{9pt}}>{\RaggedRight}p{0.25\textwidth}@{\hspace{9pt}}>{\RaggedRight}p{0.47\textwidth}@{}}
    {\bfseries #1} & {\bfseries #2} & {\bodyfont #3}
  \end{tabular}\par
  \vspace{6pt}
}

% ---------- numbered exercise item ----------
\newcommand{\eslexitem}[3]{%
  \par\vspace{2pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}l@{\hspace{9pt}}X@{}}
    {\dispfont\bfseries\footnotesize #1} &
    \RaggedRight {\bfseries #2}\par\vspace{2pt}{\color{muted}\bodyfont\small #3}
  \end{tabularx}\par
  \vspace{6pt}
}

% ---------- plain numbered question ----------
\newcommand{\eslqitem}[2]{%
  \par\vspace{2pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}l@{\hspace{9pt}}X@{}}
    {\dispfont\bfseries\footnotesize #1} &
    \RaggedRight\bodyfont #2
  \end{tabularx}\par
  \vspace{6pt}
}

% ---------- dash bullet item ----------
\newcommand{\esldash}[1]{%
  \par\vspace{2pt}%
  \noindent{\dispfont\bfseries ---}\hspace{8pt}%
  \begin{minipage}[t]{\dimexpr\textwidth-26pt\relax}\bodyfont #1\end{minipage}\par
  \vspace{6pt}
}

% ---------- accent tag ----------
\newcommand{\eslpattag}[1]{%
  \par\vspace{5pt}%
  \noindent{\bfseries\footnotesize #1}\par
  \vspace{6pt}
}

% ---------- rule head (comparatives) ----------
\newcommand{\eslrulehead}[2]{%
  \par\vspace{8pt}%
  \noindent{\dispfont\bfseries\footnotesize Rule #1}\hspace{9pt}%
  {\fontsize{11.5pt}{14pt}\selectfont\bfseries #2}\par
  \vspace{5pt}
}

% ---------- page break ----------
\newcommand{\eslpagebreak}{\clearpage}
