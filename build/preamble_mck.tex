% ---------- cover page ----------
\newcommand{\mckcover}[3]{%
  \clearpage
  \thispagestyle{empty}%
  \pagecolor{mcknavy}%
  \vspace*{42mm}%
  {\color{white}\dispfont\bfseries\fontsize{32pt}{38pt}\selectfont #1}\par
  \vspace{10pt}%
  {\color{mckmid}\fontsize{15pt}{19pt}\selectfont #2}\par
  \vspace{18pt}%
  \textcolor{mckmid}{\rule{\textwidth}{1.1pt}}\par
  \vspace{16pt}%
  {\color{white}\fontsize{12pt}{18pt}\selectfont #3}\par
  \vfill
  {\color{mckmid}\footnotesize ESLBeginner · Practical Travel English · 2026}\par
  \vspace{10mm}%
  \clearpage
  \pagecolor{white}%
}

% ---------- scenario section header (navy band) ----------
\newcommand{\mcksection}[3]{%
  \clearpage
  \par\vspace{4pt}%
  \noindent\colorbox{mcknavy}{%
    \begin{minipage}{\dimexpr\textwidth-2\fboxsep\relax}%
      \vspace{7pt}\hspace{10pt}%
      {\color{mckmid}\dispfont\bfseries\fontsize{23pt}{25pt}\selectfont #1}\hspace{12pt}%
      {\color{white}\bfseries\fontsize{15pt}{18pt}\selectfont #2}\par
      \vspace{2pt}\hspace{10pt}{\color{mckmid}\footnotesize #3}\par
      \vspace{7pt}%
    \end{minipage}}\par
  \vspace{7pt}
}

% ---------- page band (no page break) ----------
\newcommand{\mckband}[2]{%
  \par\vspace{2pt}%
  \noindent\colorbox{mcknavy}{%
    \begin{minipage}{\dimexpr\textwidth-2\fboxsep\relax}%
      \vspace{6pt}\hspace{10pt}%
      {\color{white}\bfseries\fontsize{14pt}{17pt}\selectfont #1}\hspace{10pt}%
      {\color{mckmid}\footnotesize #2}\par
      \vspace{6pt}%
    \end{minipage}}\par
  \vspace{7pt}
}

% ---------- key message (pyramid: conclusion first) ----------
\newcommand{\mckkey}[2]{%
  \par\vspace{6pt}%
  \noindent\begin{tabularx}{\textwidth}{@{}>{\columncolor{mcknavy}}p{3.2pt}@{}>{\columncolor{mcklight}}X@{}}
    & \RaggedRight {\bfseries\color{mcknavy}#1}\par\vspace{2pt}{\small #2}\par\vspace{5pt} \\
  \end{tabularx}\par
  \vspace{5pt}
}

% ---------- sub header ----------
\newcommand{\mckh}[2]{%
  \par\vspace{7pt}%
  \noindent\textcolor{mckblue}{\rule{3.2pt}{10pt}}\hspace{6pt}%
  {\bfseries\fontsize{12pt}{15pt}\selectfont #1}%
  \if\relax\detokenize{#2}\relax\else\hspace{8pt}{\color{mckgrey}\footnotesize #2}\fi\par
  \vspace{4pt}
}

% ---------- en / cn sentence pair ----------
\newcommand{\mckpair}[2]{%
  \par\vspace{2pt}%
  {\bodyfont #1}\par
  \vspace{1.5pt}
  {\color{mckgrey}\footnotesize #2}\par
  \vspace{4pt}
}

% ---------- plain note ----------
\newcommand{\mcknote}[1]{%
  \par\vspace{2pt}{\color{mckgrey}\small #1}\par\vspace{5pt}
}

% ---------- tip ----------
\newcommand{\mcktip}[2]{%
  \par\vspace{4pt}%
  \noindent{\color{mckblue}\dispfont\bfseries\footnotesize TIP}\hspace{7pt}%
  {\bfseries #1}\hspace{7pt}{\color{mckgrey}\small #2}\par\vspace{6pt}
}

% ---------- word / phrase row ----------
\newcommand{\mckword}[2]{%
  \par\vspace{2pt}%
  \noindent{\bfseries #1}\hspace{8pt}{\color{mckgrey}#2}\par\vspace{5pt}
}
