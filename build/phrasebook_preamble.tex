% ---------- entries ----------
% Every cell is \smash-ed, so the row keeps a fixed, predictable height whatever
% a cell contains (the CJK glyph box is ~1.45em tall and would otherwise push
% each row to ~20pt, and a long cell would silently push the row below it).
% The row advance is therefore stated explicitly by the caller:
%     advance = 2.2pt + 13pt * (extra lines - 1)
% which is 14.2pt for a normal single-line row, so every entry keeps the same
% rhythm. #1 is the row's font size: the builder lowers it (never below 8.25pt)
% for the few rows whose longest cell is wider than its column, which is what
% keeps an entry on one line instead of wrapping it.
%
% Every cell opens with \pbstrut so that all four columns share one baseline:
% the cells are top-aligned, so without it the first line of each cell would
% sit at top + its own font's ascent (CJK ascent is ~1.5pt taller than the
% Latin one) and the row would look ragged. The strut (12pt) is taller than
% any of the four ascents, so top + 12pt is the baseline everywhere.
\newcommand{\pbstrut}{\rule{0pt}{12pt}}
\newcommand{\pbRowCore}[5]{%
  \smash{\parbox[t]{\pbIdxW}{\pbstrut\raggedleft\color{faint}\fontsize{8pt}{13pt}\selectfont #2}} &
  \smash{\parbox[t]{\pbPhraseW}{\pbstrut\raggedright\fontsize{#1pt}{13pt}\selectfont\bfseries #3}} &
  \smash{\parbox[t]{\pbGlossW}{\pbstrut\raggedright\color{muted}\fontsize{#1pt}{13pt}\selectfont #4}} &
  \smash{\parbox[t]{\pbExW}{\pbstrut\raggedright\color{muted}\fontsize{#1pt}{13pt}\selectfont\itshape #5}}}

% \phentry{size}{space after the row}{index}{phrase}{gloss}{example}
\newcommand{\phentry}[6]{\pbRowCore{#1}{#3}{#4}{#5}{#6}\\[#2pt]}
% "keep" is used on the second-to-last row of a section: it forbids the page
% break that would leave the section's last row alone at the top of a page.
% The star form still takes the row space, so the rhythm stays uniform.
\newcommand{\phentrykeep}[6]{\pbRowCore{#1}{#3}{#4}{#5}{#6}\\*[#2pt]}
