% ============================================================
%  ESLBeginner · Beginner series addendum (loaded after preamble.tex)
%
%  The shared preamble's \eslsection puts its number in Montserrat, which has
%  no CJK glyphs — fine for the older worksheets (Latin section titles) but not
%  for the Beginner series, whose headings are mostly Chinese. These variants
%  keep the same visual rhythm with the CJK-capable main font instead.
% ============================================================

% ---------- page ----------
% Narrower side margins than the shared 20mm: the Beginner tables carry a
% bilingual column alongside the example, and the extra width is what keeps
% every cell on a single line.
\geometry{left=14mm, right=14mm}

% ---------- vertical rhythm ----------
% The series carries no exercises, so a lesson is knowledge tables only and
% needs a looser setting than the old 12.6pt to fill its two A4 pages.
\setlength{\baselineskip}{14.4pt}
\geometry{top=15mm, bottom=15mm}

% ---------- keep headings with their content ----------
% Without this a section heading can land alone at the foot of a page with its
% table overleaf. \needspace moves the heading instead of orphaning it.
\usepackage{needspace}

% Tables are longtables so they can break across a page instead of moving over
% whole — an unbreakable tabular strands its heading at the foot of the page.
\usepackage{longtable}

% ---------- two spare lines on the page a table ends on ----------
% A table that reaches the foot of page one spills its remainder onto page two.
% When exactly one row spills, that row reads as a stray line: longtable ships
% a table's final page through its end-of-table branch, which does not repeat
% the header, so the row has neither header nor neighbour (05, 06, 28). The
% break point inside a longtable cannot be steered from outside it, so give the
% page two spare lines instead: a table that was a row short now fits, and
% nothing else moves.
%
% \vsize is the register that moves the break. TeX's page builder fills to
% \pagegoal, which it initialises from \vsize, and \@colht/\@colroom are
% LaTeX's account of the same thing — but longtable only copies \@colroom back
% into \vsize itself once it has already shipped a page. A table starting near
% the foot of page one therefore measured against the bare \textheight and
% spilled its last row however \@colht was set; setting \vsize directly is what
% changes where it breaks.
%
% \enlargethispage ADDS to the page goal, so issuing it before every table
% compounds — on a table-dense page the goal grew by half a page and content
% ran into the bottom margin. This does the same job but SETS the goal, so
% repeated calls are idempotent whatever the page holds. It has to be issued
% on the page being lengthened, which is why the builder calls it before each
% table rather than once at the top.
\makeatletter
\newcommand{\eslspareline}{%
  \global\@colht\textheight
  \global\advance\@colht by 2\baselineskip
  \global\@colroom\@colht
  \global\vsize\@colht
  \pagegoal\@colroom}
\makeatother

% ---------- document title ----------
\newcommand{\eslbtitle}[1]{%
  \par\vspace{0pt}%
  {\bfseries\fontsize{22pt}{26pt}\selectfont #1}\par
  \vspace{14pt}%
}

% ---------- numbered section heading (number in Montserrat, text CJK-safe) ----------
\newcommand{\eslbsec}[2]{%
  \needspace{4\baselineskip}%
  \par\vspace{13pt}%
  \noindent{\dispfont\bfseries\fontsize{14pt}{17pt}\selectfont #1}\hspace{8pt}%
  {\bfseries\fontsize{12.5pt}{16pt}\selectfont #2}\par
  \vspace{7pt}%
}

% ---------- sub-heading (### in the source) ----------
\newcommand{\eslbsub}[1]{%
  \needspace{3\baselineskip}%
  \par\vspace{8pt}%
  \noindent{\bfseries\small #1}\par
  \vspace{4pt}%
}
